    local tgl = Instance.new("TextButton")

    tgl.BackgroundColor3 = Color3.fromRGB(30, 30, 36)
    tgl.Text = "  " .. text
    tgl.TextColor3 = Color3.fromRGB(220, 220, 220)
    tgl.TextSize = 13

    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = tgl
    
    local status = Instance.new("TextLabel")
    status.Size = UDim2.new(0, 60, 1, 0)
    status.Position = UDim2.new(1, -70, 0, 0)
    status.BackgroundTransparency = 1
    status.Text = default and "[ON]" or "[OFF]"
    status.TextColor3 = default and Color3.fromRGB(50, 255, 50) or Color3.fromRGB(255, 50, 50)
    status.TextSize = 13
    status.Font = Enum.Font.GothamBold
    status.Parent = tgl
    
    local state = default
    tgl.MouseButton1Click:Connect(function()
        state = not state
        status.Text = state and "[ON]" or "[OFF]"
        status.TextColor3 = state and Color3.fromRGB(50, 255, 50) or Color3.fromRGB(255, 50, 50)
        callback(state)
    end)
end

local function giveItem(itemName)
    local itemGiver = Workspace:FindFirstChild("Prison_ITEMS") and Workspace.Prison_ITEMS:FindFirstChild("giver") and Workspace.Prison_ITEMS.giver:FindFirstChild(itemName)

        local itemHandler = ReplicatedStorage:FindFirstChild("ItemHandler")
        if pickupPart and itemHandler then
            itemHandler:InvokeServer(pickupPart)
        end
    end
end

local function getPlayerColor(player)
    local team = player.Team and player.Team.Name or ""
    if team == "Guards" then return Color3.fromRGB(50, 120, 255)
    else return Color3.fromRGB(255, 40, 40) end
end

local function isEnemy(player)
    if player == LocalPlayer then return false end
    local myTeam = LocalPlayer.Team and LocalPlayer.Team.Name or ""
    local targetTeam = player.Team and player.Team.Name or ""
    if myTeam == "Guards" then return targetTeam == "Inmates" or targetTeam == "Criminals"
    elseif myTeam == "Inmates" then return targetTeam == "Guards" or targetTeam == "Criminals"
    elseif myTeam == "Criminals" then return targetTeam == "Guards" or targetTeam == "Inmates" end
    return false
end

-- ESP
local espHighlights = {}
RunService.RenderStepped:Connect(function()
    if not getgenv().PL_Settings.ESP_Enabled then
        for _, hl in pairs(espHighlights) do hl:Destroy() end
        espHighlights = {}
        return
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if isEnemy(player) then
            if player.Character and not espHighlights[player] then
                local highlight = Instance.new("Highlight")
                highlight.Name = "Liwys_ESP"
                highlight.Adornee = player.Character
                highlight.FillColor = getPlayerColor(player)
                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                highlight.FillTransparency = 0.4
                highlight.Parent = player.Character
                espHighlights[player] = highlight
            elseif player.Character and espHighlights[player] then
                espHighlights[player].FillColor = getPlayerColor(player)
            elseif not player.Character and espHighlights[player] then
                espHighlights[player]:Destroy()
                espHighlights[player] = nil
            end
        else
            if espHighlights[player] then
                espHighlights[player]:Destroy()
                espHighlights[player] = nil
            end
        end
    end
end)

local function showDamage(targetPart, damageAmount)
    if not getgenv().PL_Settings.DamageIndicator or not targetPart then return end
    local billboard = Instance.new("BillboardGui")
    billboard.Size = UDim2.new(0, 60, 0, 40)
    billboard.StudsOffset = Vector3.new(math.random(-1, 1), 3, math.random(-1, 1))
    billboard.Adornee = targetPart
    
    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.new(1, 0, 1, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.Text = "-" .. tostring(damageAmount)
    textLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
    textLabel.TextSize = 20
    textLabel.Font = Enum.Font.GothamBold
    textLabel.TextStrokeTransparency = 0
    textLabel.Parent = billboard
    
    billboard.Parent = targetPart
    game:GetService("Debris"):AddItem(billboard, 0.7)
end

local function hookPlayerHealth(player)
    if player == LocalPlayer then return end
    local function setupCharacter(char)
        local humanoid = char:WaitForChild("Humanoid", 5)
        if humanoid then
            local lastHealth = humanoid.Health
            humanoid.HealthChanged:Connect(function(newHealth)
                if newHealth < lastHealth and isEnemy(player) then
                    local damage = math.floor(lastHealth - newHealth)
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    if hrp and damage > 0 then showDamage(hrp, damage) end
                end
                lastHealth = newHealth
            end)
        end
    end
    if player.Character then setupCharacter(player.Character) end
    player.CharacterAdded:Connect(setupCharacter)
end

for _, p in ipairs(Players:GetPlayers()) do hookPlayerHealth(p) end
Players.PlayerAdded:Connect(hookPlayerHealth)

-- Аимбот
local isHoldingRMB = false
UserInputService.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then isHoldingRMB = true end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then isHoldingRMB = false end
end)

RunService.RenderStepped:Connect(function()
    if getgenv().PL_Settings.Aimbot_Enabled and isHoldingRMB then
        local closestTarget = nil
        local shortestDist = getgenv().PL_Settings.FOV_Radius
        local viewportCenter = Camera.ViewportSize / 2
        
        for _, player in ipairs(Players:GetPlayers()) do
            if isEnemy(player) and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
                local partName = getgenv().PL_Settings.AimTargetPart
                local targetPart = player.Character:FindFirstChild(partName)
                if targetPart then
                    local screenPoint, onScreen = Camera:WorldToScreenPoint(targetPart.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPoint.X, screenPoint.Y) - viewportCenter).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestTarget = targetPart
                        end
                    end
                end
            end
        end
        
        if closestTarget then
